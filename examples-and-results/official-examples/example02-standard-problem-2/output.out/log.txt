//output directory: output.out/
Msat = 1000e3
Aex = 10e-12
lex := sqrt(10e-12 / (0.5 * mu0 * pow(1000e3, 2)))
d := 30 * lex
Sizex := 5 * d
Sizey := 1 * d
Sizez := 0.1 * d
nx := pow(2, ilogb(Sizex/(0.75*lex)))
ny := pow(2, ilogb(Sizey/(0.75*lex)))
SetGridSize(nx, ny, 1)
SetCellSize(Sizex/nx, Sizey/ny, Sizez)
m = Uniform(1, 0.1, 0)
relax()
save(m)
print("<m> for d/lex=30: ", m.average())
//<m> for d/lex=30:  [0.9630156755447388 0.07534648478031158 0]
