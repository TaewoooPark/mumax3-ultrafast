setgridsize(N, N, 1)

setcellsize(c, c, c)
setgeom(circle(N * c))

defregion(2, xrange(-inf, 0))
save(regions)

anisU.setRegion(1, vector(1, 0, 0))

anisU.setRegion(2, vector(0, 1, 0))
save(Ku1)
save(anisU)

save(Msat)

alpha = 1

m.setRegion(2, uniform(-1, 1, 0))
saveas(m, "m_inital")

saveas(m, "m_final")
