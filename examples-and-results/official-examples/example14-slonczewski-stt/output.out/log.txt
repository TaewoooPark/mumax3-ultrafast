//output directory: output.out/
sizeX := 160e-9
sizeY := 80e-9
sizeZ := 5e-9
Nx := 64
Ny := 32
setgridsize(Nx, Ny, 1)
setcellsize(sizeX/Nx, sizeY/Ny, sizeZ)
setGeom(ellipse(sizeX, sizeY))
Msat = 800e3
Aex = 13e-12
alpha = 0.01
m = uniform(1, 0, 0)
lambda = 1
Pol = 0.5669
epsilonprime = 0
angle := 20
px := cos(angle * pi / 180)
py := sin(angle * pi / 180)
fixedlayer = vector(px, py, 0)
Jtot := -0.008
area := sizeX * sizeY * pi / 4
jc := Jtot / area
J = vector(0, 0, jc)
autosave(m, 100e-12)
tableautosave(10e-12)
run(1e-9)
